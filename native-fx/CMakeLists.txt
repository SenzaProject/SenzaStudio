cmake_minimum_required(VERSION 3.18)
project(senzastudiofx CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)

# The shared library loaded by audioserver / the effect HAL. Name is deliberately NOT "libsenzafx"
# so it never collides with the SenzaSound library of the same name.
add_library(senzastudiofx SHARED src/senzafx_dsp.cpp src/senzafx_effect.cpp)
target_include_directories(senzastudiofx PRIVATE compat)
target_compile_options(senzastudiofx PRIVATE -O3 -fvisibility=hidden -fno-exceptions -fno-rtti -Wall -Wextra -Werror)
set_target_properties(senzastudiofx PROPERTIES OUTPUT_NAME senzastudiofx)
option(SENZAFX_SMOKETEST "Start enabled with +9 dB sub-bass (audible smoke test)" OFF)
if(SENZAFX_SMOKETEST)
  target_compile_definitions(senzastudiofx PRIVATE SENZAFX_SMOKETEST=1)
  set_target_properties(senzastudiofx PROPERTIES OUTPUT_NAME senzastudiofx)
endif()
if(ANDROID)
  # 16 KB page-size compatible, no undefined symbols, no hidden dependency on libc++_shared.
  target_link_options(senzastudiofx PRIVATE -Wl,--no-undefined -Wl,-z,max-page-size=16384 -Wl,--gc-sections)
endif()

# Host-only tests (not built for Android):  cmake -B build-host && cmake --build build-host && ctest --test-dir build-host
if(NOT ANDROID)
  enable_testing()
  find_package(Threads REQUIRED)
  foreach(t test_dsp test_effect test_race)
    add_executable(${t} test/${t}.cpp src/senzafx_dsp.cpp src/senzafx_effect.cpp)
    target_include_directories(${t} PRIVATE compat)
    target_compile_options(${t} PRIVATE -O1 -g)
    target_link_libraries(${t} PRIVATE Threads::Threads)
    add_test(NAME ${t} COMMAND ${t})
  endforeach()
  target_compile_options(test_dsp PRIVATE -fsanitize=address,undefined -fno-sanitize-recover=undefined)
  target_link_options(test_dsp PRIVATE -fsanitize=address,undefined)
  target_compile_options(test_effect PRIVATE -fsanitize=address,undefined -fno-sanitize-recover=undefined)
  target_link_options(test_effect PRIVATE -fsanitize=address,undefined)
  target_compile_options(test_race PRIVATE -fsanitize=thread)
  target_link_options(test_race PRIVATE -fsanitize=thread)
endif()
